name: Publish here.now

on:
  workflow_dispatch:
    inputs:
      tag:
        description: ottplay-foss release tag (empty = latest)
        required: false
        default: ""
  repository_dispatch:
    types: [ottplay-foss-release]
  release:
    types: [published]

concurrency:
  group: herenow-player
  cancel-in-progress: true

jobs:
  publish:
    runs-on: ubuntu-latest
    permissions:
      contents: read
    steps:
      - name: Checkout vitrine
        uses: actions/checkout@v4

      - name: Publish gate
        id: gate
        env:
          HERENOW_API_KEY: ${{ secrets.HERENOW_API_KEY }}
        run: |
          if [ -z "$HERENOW_API_KEY" ]; then
            echo "missing=1" >> "$GITHUB_OUTPUT"
            echo "HERENOW_API_KEY not set — skip publish."
          else
            echo "missing=0" >> "$GITHUB_OUTPUT"
          fi

      - name: Resolve ottplay-foss tag
        if: steps.gate.outputs.missing == '0'
        id: tag
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          T="${{ github.event.client_payload.tag || inputs.tag }}"
          if [ -z "$T" ]; then
            T=$(gh api repos/open-ott-play/ottplay-foss/releases/latest --jq .tag_name)
          fi
          echo "tag=$T" >> "$GITHUB_OUTPUT"
          echo "Using ottplay-foss $T"

      - name: Download + unpack dist
        if: steps.gate.outputs.missing == '0'
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          set -euo pipefail
          TAG="${{ steps.tag.outputs.tag }}"
          gh release download "$TAG" -R open-ott-play/ottplay-foss -p ottplay-foss-dist.tar.gz
          rm -rf dist staging
          mkdir -p staging
          tar -xzf ottplay-foss-dist.tar.gz -C staging
          if [ -f staging/index.html ]; then
            mv staging dist
          elif [ -f staging/dist/index.html ]; then
            mv staging/dist dist
          else
            top=$(find staging -mindepth 1 -maxdepth 1 -type d | head -1)
            if [ -n "$top" ] && [ -f "$top/index.html" ]; then
              mv "$top" dist
            else
              echo "Could not find index.html:" >&2
              find staging -maxdepth 3 -type f | head -50 >&2
              exit 1
            fi
          fi
          test -f dist/index.html

      - name: Setup Node
        if: steps.gate.outputs.missing == '0'
        uses: actions/setup-node@v4
        with:
          node-version: "22"

      - name: Install here.now + publish
        if: steps.gate.outputs.missing == '0'
        env:
          HERENOW_API_KEY: ${{ secrets.HERENOW_API_KEY }}
          HERENOW_WORKSPACE: ottplay
          HERENOW_SITE_SLUG: liminal-sketch-vv8r
          OVERWRITE: "1"
          SPA: "1"
        run: |
          set -euo pipefail
          curl -fsSL https://here.now/install.sh | bash
          PUBLISH=""
          for c in \
            "$HOME/.claude/skills/here-now/scripts/publish.sh" \
            "$HOME/.agents/skills/here-now/scripts/publish.sh" \
            "$HOME/.hermes/skills/here-now/scripts/publish.sh" \
            "$HOME/.cursor/skills/here-now/scripts/publish.sh" \
            ".cursor/skills/here-now/scripts/publish.sh"
          do
            if [ -x "$c" ]; then PUBLISH="$c"; break; fi
          done
          if [ -z "$PUBLISH" ]; then
            PUBLISH=$(find "$HOME" -path '*here-now*/scripts/publish.sh' 2>/dev/null | head -1 || true)
            if [ -n "$PUBLISH" ]; then chmod +x "$PUBLISH" || true; fi
          fi
          if [ -z "$PUBLISH" ] || [ ! -x "$PUBLISH" ]; then
            echo "publish.sh not found after install" >&2
            find "$HOME" -path '*here-now*/publish.sh' 2>/dev/null | head -20 >&2 || true
            exit 1
          fi
          export HERENOW_PUBLISH_SCRIPT="$PUBLISH"
          chmod +x scripts/publish-herenow.sh
          if ! ./scripts/publish-herenow.sh ./dist; then
            echo "publish failed once — retrying after 20s" >&2
            sleep 20
            if ! ./scripts/publish-herenow.sh ./dist; then
              echo "publish failed twice — checking live player URL" >&2
              code=$(curl -sS -o /dev/null -w '%{http_code}' https://player.ottplay.here.now/ || echo 000)
              echo "player.ottplay.here.now → HTTP $code" >&2
              if [ "$code" != "200" ]; then
                exit 1
              fi
              echo "live URL still 200; treating as soft success (likely finalize response glitch)" >&2
            fi
          fi
